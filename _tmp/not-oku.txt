// K-900: atlas-not-oku — elle doldurulmuş soru-puan çizelgesinin (karekodlu Atlas kâğıdı) puan tablosunu okur.
// Gelen görüntüde öğrenci numarası ve adı panelde BOYANMIŞTIR; yalnız sıra numarası, soru puanları ve toplam görünür.
// Anahtar yalnız burada: Supabase Secrets → GEMINI_API_KEY.
const SUPA_URL = Deno.env.get("SUPABASE_URL")!;
const ANON = Deno.env.get("SUPABASE_ANON_KEY") ?? "";
const MODEL = Deno.env.get("ATLAS_OKU_MODEL") ?? "gemini-flash-latest";
const YEDEK = Deno.env.get("ATLAS_YEDEK_MODEL") ?? "gemini-flash-lite-latest";
const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info", "Access-Control-Allow-Methods": "POST, OPTIONS" };
const cevap = (o: unknown, s = 200) => new Response(JSON.stringify(o), { status: s, headers: { ...CORS, "Content-Type": "application/json; charset=utf-8" } });

const talimat = (n: number, q: number) => `Görüntü bir okulun sınav puan çizelgesidir (A4 kâğıdın düzeltilmiş tablo bölümü).
Sütunlar soldan sağa: "Sıra" (basılı 1..${n}), "No" ve "Adı Soyadı" (gizlilik için gri boyalı; okuma), ${q} soru sütunu (başlıkta 1..${q}), en sağda "Toplam".
Başlığın hemen altındaki ilk satır "Soru puanı" satırıdır: her sorunun en yüksek puanı (elle yazılır, boş olabilir).
Sonraki ${n} satır öğrencilerdir (sıra 1..${n}). Kutulara öğretmen ELLE puan yazmıştır; "Toplam" kutusunda toplam puan ya da G (sınava girmedi) veya K (kopya) olabilir.
Kurallar:
- Her kutuyu ayrı oku; gördüğünü yaz, hesaplayarak düzeltme ya da tahmin YAPMA. Toplamı soruların toplamından türetme; kutuda yazanı yaz.
- Boş kutu = null. Yarım puan "2,5" ya da "2.5" olabilir → 2.5.
- Üstü çizilmiş rakam varsa yanındaki/üstündeki son yazılanı oku.
- Emin olmadığın kutunun sütun numarasını (1..${q}; toplam için 0) o satırın "belirsiz" listesine ekle.
- Tam ${n} öğrenci satırı döndür, sıra sırasıyla. puan dizisi tam ${q} eleman.`;
const sema = (q: number) => ({
  type: "OBJECT",
  properties: {
    maxlar: { type: "ARRAY", items: { type: "NUMBER", nullable: true } },
    satirlar: { type: "ARRAY", items: { type: "OBJECT", properties: {
      s: { type: "INTEGER" }, puan: { type: "ARRAY", items: { type: "NUMBER", nullable: true } }, toplam: { type: "STRING" }, belirsiz: { type: "ARRAY", items: { type: "INTEGER" } } }, required: ["s", "puan", "toplam"] } },
  },
  required: ["maxlar", "satirlar"],
});

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  if (req.method !== "POST") return cevap({ hata: "Yalnız POST" }, 405);
  const yetki = req.headers.get("Authorization") ?? "";
  const rol = await fetch(`${SUPA_URL}/rest/v1/rpc/umal_benim_rolum`, { method: "POST", headers: { apikey: ANON, Authorization: yetki, "Content-Type": "application/json" }, body: "{}" });
  const r = rol.ok ? await rol.json() : null;
  if (!["bas", "yonetici", "rehberlik", "ogretmen"].includes(r)) return cevap({ hata: "Yetkisiz" }, 401);
  let g: { resim?: string; satir?: number; soru?: number };
  try { g = await req.json(); } catch { return cevap({ hata: "Geçersiz istek" }, 400); }
  const resim = typeof g.resim === "string" && g.resim.length > 1000 && g.resim.length < 8_000_000 ? g.resim : "";
  const n = Math.max(1, Math.min(40, Math.round(Number(g.satir) || 0))), q = Math.max(1, Math.min(12, Math.round(Number(g.soru) || 10)));
  if (!resim) return cevap({ hata: "Görüntü yok" }, 400);
  const anahtar = Deno.env.get("GEMINI_API_KEY");
  if (!anahtar) return cevap({ hata: "Yapay zekâ anahtarı kurulmamış" }, 503);
  const govde = (s: boolean, dusun: boolean) => JSON.stringify({
    systemInstruction: { parts: [{ text: talimat(n, q) + (s ? "" : `\nYalnız JSON döndür: {"maxlar":[...${q}],"satirlar":[{"s":1,"puan":[...${q}],"toplam":"","belirsiz":[]}]}`) }] },
    contents: [{ role: "user", parts: [{ text: `Bu çizelgedeki ${n} öğrenci satırını ve soru puanı satırını oku.` }, { inline_data: { mime_type: "image/jpeg", data: resim } }] }],
    generationConfig: { temperature: 0, maxOutputTokens: 16000, responseMimeType: "application/json", ...(s ? { responseSchema: sema(q) } : {}), ...(dusun ? { thinkingConfig: { thinkingLevel: "low" } } : {}) },
  });
  const ayikla = (y: string) => { const t = y.trim().replace(/^```(?:json)?\s*/i, "").replace(/```\s*$/, ""); const a = t.indexOf("{"), b = t.lastIndexOf("}"); return JSON.parse(a >= 0 && b > a ? t.slice(a, b + 1) : t); };
  const sayi = (v: unknown) => { if (v == null || v === "") return null; const x = Number(String(v).replace(",", ".")); return Number.isFinite(x) && x >= 0 && x <= 100 ? Math.round(x * 100) / 100 : null; };
  let son = 0, ayrinti = "";
  for (const m of [MODEL, MODEL, YEDEK]) {
    const istek = (s: boolean, d: boolean) => fetch(`https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(m)}:generateContent`, { method: "POST", headers: { "x-goog-api-key": anahtar, "Content-Type": "application/json" }, body: govde(s, d) });
    let yz = await istek(true, true);
    if (yz.status === 400) yz = await istek(true, false);
    if (yz.status === 400) yz = await istek(false, false);
    son = yz.status;
    if (!yz.ok) { ayrinti = (await yz.text()).slice(0, 300); continue; }
    const j = await yz.json();
    const yazi = (j?.candidates?.[0]?.content?.parts ?? []).filter((p: { thought?: boolean }) => !p.thought).map((p: { text?: string }) => p.text ?? "").join("");
    try {
      const o = ayikla(yazi);
      const maxlar = Array.from({ length: q }, (_, i) => sayi((o.maxlar ?? [])[i]));
      const S = Array.isArray(o.satirlar) ? o.satirlar : [];
      const satirlar = Array.from({ length: n }, (_, i) => {
        const x = S.find((z: { s?: number }) => Number(z?.s) === i + 1) ?? S[i] ?? {};
        const top = String(x.toplam ?? "").trim().toUpperCase().replace(/\s+/g, "");
        return { s: i + 1, puan: Array.from({ length: q }, (_, k) => sayi((x.puan ?? [])[k])), toplam: /^(G|K)$/.test(top) ? top : (sayi(top) == null ? (top ? "?" : "") : String(sayi(top))),
          belirsiz: (Array.isArray(x.belirsiz) ? x.belirsiz : []).map(Number).filter((v: number) => v >= 0 && v <= q) };
      });
      return cevap({ ok: true, model: m, maxlar, satirlar });
    } catch { ayrinti = "JSON çözülemedi"; }
  }
  return cevap({ ok: false, hata: "Puanlar okunamadı (" + son + ")", ayrinti }, 502);
});
